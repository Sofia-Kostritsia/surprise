#!/usr/bin/env python3 # -*- coding: utf-8 -*- """ Excel Comment Statistics ------------------------ Збирає проблеми з колонок "Коментар" у великій кількості Excel-файлів. Запуск: python excel_comment_statistics.py Потрібно: pip install openpyxl """ import os import re import sys import traceback from collections import Counter from pathlib import Path from datetime import datetime try: import tkinter as tk from tkinter import filedialog, messagebox except Exception: tk = None from openpyxl import load_workbook, Workbook from openpyxl.styles import Font, PatternFill, Alignment, Border, Side from openpyxl.utils import get_column_letter IGNORE_SHEETS = {"logs"} # Тут можна додавати власні категорії. # Категорія спрацьовує, якщо будь-яке ключове слово є в коментарі. CATEGORIES = { # Фізичні пошкодження деталей/корпусу. "Вирівнювання столу / Bed leveling": [ "height map", "heightmap", "регулювальн", "регулююч", "шайб", "вирівнюв", "рівнян", "нахил стол", "bed leveling", "bed level", "leveling", "level" ], # Калібрування геометрії/висоти інструмента. "Калібрування / Tool Height": [ "калібр", "калібров", "tool height", "інструментальн висот", "offset", "calibration" ], # Рухова частина. "Двигуни / рух / механіка": [ "двигун", "мотор", "motor", "вібрац", "вібра", "закуш", "люфт", "рух", "кінцевик", "endstop" ], # Нагрів, температурні датчики та вентилятори. "Температура / нагрів": [ "нагрів", "нагріва", "температур", "heater", "heating", "термодат", "терміст", "thermistor" ], "Вентилятори": [ "вентилят", "fan", "hepa" ], # Друк та якість надрукованої деталі. "Якість друку": [ "друк", "друку", "шар", "пластик", "сопл", "екструзі", "vfa", "print", "layer", "nozzle", "extrud" ], # Живлення та електроніка, якщо проблема не стосується конкретно кабелю/CAN. "Електроніка / живлення": [ "живлен", "напруг", "струм", "електр", "duet", "24v", "110v", "220v", "wifi", "wi-fi", "плата" ], "Забруднення / сторонні матеріали": [ "бруд", "забруд", "пил", "залиш", "слід", "філамент" ], } def clean(value): if value is None: return "" return str(value).strip() def normalize_text(value): text = clean(value).lower() text = text.replace("’", "'").replace("`", "'") text = re.sub(r"\s+", " ", text) return text.strip() def normalize_comment(value): text = normalize_text(value) # Прибираємо незначні відмінності для підрахунку повторів. text = re.sub(r"[.,;:!?]+$", "", text) return text def classify_comment(comment): text = normalize_text(comment) # Specific rules first. if any(x in text for x in ["can-bus", "can bus", "canbus"]): return "Електрика / проводка / CAN" if any(x in text for x in ["регулювальн", "регулююч", "шайб", "height map", "heightmap", "bed leveling", "bed level", "вирівнюв", "рівнян"]): return "Вирівнювання столу / Bed leveling" if any(x in text for x in ["tool height", "калібр", "калібров", "calibration", "offset"]): return "Калібрування / Tool Height" if any(x in text for x in ["подряп", "пошкод", "дефект", "тріс", "скол", "вм'ят", "вмят", "царап", "scratch", "damage", "damaged"]): return "Механічні пошкодження" # Then use the remaining categories. for category, keywords in CATEGORIES.items(): if any(keyword.lower() in text for keyword in keywords): return category return "Інше" def find_value_near_label(ws, label): label_norm = normalize_text(label).rstrip(":") for row in ws.iter_rows(): for cell in row: if normalize_text(cell.value).rstrip(":") == label_norm: # Найчастіше значення праворуч. for c in range(cell.column + 1, min(ws.max_column, cell.column + 4) + 1): value = clean(ws.cell(cell.row, c).value) if value: return value return "" def find_comment_header(ws): """ Повертає (row, column) для заголовка Коментар. Якщо є кілька заголовків, беремо той, який належить основній таблиці. """ candidates = [] for row in ws.iter_rows(): for cell in row: if normalize_text(cell.value) in {"коментар", "коментарі"}: candidates.append((cell.row, cell.column, cell.value)) if not candidates: return None # Перевага колонці 9 та заголовку "Коментар". candidates.sort(key=lambda x: ( 0 if x[2] == "Коментар" else 1, abs(x[1] - 9), x[0] )) return candidates[0][0], candidates[0][1] def find_table_header(ws): """ Шукає рядок основної таблиці за "Група тестів", "Назва тесту", "Статус". """ for r in range(1, ws.max_row + 1): values = {normalize_text(ws.cell(r, c).value) for c in range(1, ws.max_column + 1)} if "назва тесту" in values and "статус" in values: headers = {} for c in range(1, ws.max_column + 1): v = normalize_text(ws.cell(r, c).value) if v: headers[v] = c return r, headers return None, {} def extract_sheet(ws, filename): if normalize_text(ws.title) in {normalize_text(x) for x in IGNORE_SHEETS}: return [], [] serial = find_value_near_label(ws, "Серійний номер") stage = find_value_near_label(ws, "Етап") machines = [] if serial: machines.append({ "Файл": filename, "Серійний номер": serial, "Етап": stage, "Аркуш": ws.title, }) header_row, headers = find_table_header(ws) comment_header = find_comment_header(ws) if not comment_header: return machines, [] comment_row, comment_col = comment_header records = [] main_comment_col = headers.get("коментар") if header_row and main_comment_col: for r in range(header_row + 1, ws.max_row + 1): comment = clean(ws.cell(r, main_comment_col).value) test_name_col = headers.get("назва тесту") test_number_col = headers.get("номер") group_col = headers.get("група тестів") status_col = headers.get("статус") test_name = clean(ws.cell(r, test_name_col).value) if test_name_col else "" test_number = clean(ws.cell(r, test_number_col).value) if test_number_col else "" group = clean(ws.cell(r, group_col).value) if group_col else "" status = clean(ws.cell(r, status_col).value) if status_col else "" if comment and normalize_text(comment) not in {"коментар", "коментарі"}: records.append({ "Файл": filename, "Серійний номер": serial, "Аркуш": ws.title, "Етап": stage, "Група тестів": group, "№ тесту": test_number, "Назва тесту": test_name, "Статус": status, "Коментар": comment, "Категорія": classify_comment(comment), }) else: for r in range(comment_row + 1, ws.max_row + 1): comment = clean(ws.cell(r, comment_col).value) if not comment or normalize_text(comment) in {"коментар", "коментарі"}: continue records.append({ "Файл": filename, "Серійний номер": serial, "Аркуш": ws.title, "Етап": stage, "Група тестів": "", "№ тесту": "", "Назва тесту": "", "Статус": "", "Коментар": comment, "Категорія": classify_comment(comment), }) return machines, records def process_folder(folder): folder = Path(folder) all_records = [] all_machines = [] errors = [] files = sorted( p for p in folder.iterdir() if p.is_file() and p.suffix.lower() in {".xlsx", ".xlsm"} and not p.name.startswith("~$") ) for file_path in files: try: wb = load_workbook(file_path, data_only=True, read_only=False) for ws in wb.worksheets: try: machines, records = extract_sheet(ws, file_path.name) all_machines.extend(machines) all_records.extend(records) except Exception as exc: errors.append((file_path.name, ws.title, str(exc))) wb.close() except Exception as exc: errors.append((file_path.name, "", str(exc))) # One machine per serial number. unique_machines = {} for m in all_machines: serial = clean(m.get("Серійний номер")) if serial and serial not in unique_machines: unique_machines[serial] = m # Remove duplicates from backup/duplicate Excel files. # Same serial + sheet + group + test + comment = one problem. unique_records = {} for r in all_records: key = ( normalize_text(r.get("Серійний номер")), normalize_text(r.get("Аркуш")), normalize_text(r.get("Група тестів")), normalize_text(r.get("№ тесту")), normalize_text(r.get("Назва тесту")), normalize_comment(r.get("Коментар")), ) if key not in unique_records: unique_records[key] = r return list(unique_records.values()), list(unique_machines.values()), errors, files def autosize(ws, max_width=55): for col_cells in ws.columns: length = 0 col_letter = get_column_letter(col_cells[0].column) for cell in col_cells: value = clean(cell.value) if value: length = max(length, max(len(line) for line in value.split("\n"))) ws.column_dimensions[col_letter].width = min(max(length + 2, 10), max_width) def style_sheet(ws): header_fill = PatternFill("solid", fgColor="1F4E78") header_font = Font(color="FFFFFF", bold=True) thin = Side(style="thin", color="D9E1F2") for cell in ws[1]: if cell.value is not None: cell.fill = header_fill cell.font = header_font cell.alignment = Alignment(horizontal="center", vertical="center", wrap_text=True) for row in ws.iter_rows(): for cell in row: cell.alignment = Alignment(vertical="top", wrap_text=True) cell.border = Border(bottom=thin) ws.freeze_panes = "A2" ws.auto_filter.ref = ws.dimensions autosize(ws) def write_report(records, machines, errors, files, output_path): wb = Workbook() ws = wb.active ws.title = "Усі проблеми" columns = [ "Файл", "Серійний номер", "Аркуш", "Етап", "Група тестів", "№ тесту", "Назва тесту", "Статус", "Категорія", "Коментар" ] for c, name in enumerate(columns, 1): ws.cell(1, c, name) records_by_serial = {} for record in records: serial = clean(record.get("Серійний номер")) or "(серійний номер не знайдено)" records_by_serial.setdefault(serial, []).append(record) all_serials = [] for m in machines: serial = clean(m.get("Серійний номер")) if serial and serial not in all_serials: all_serials.append(serial) for serial in records_by_serial: if serial not in all_serials: all_serials.append(serial) row = 2 serial_ranges = [] for serial in all_serials: serial_records = records_by_serial.get(serial, []) if serial_records: start_row = row for record in serial_records: for c, name in enumerate(columns, 1): ws.cell(row, c, record.get(name, "")) row += 1 end_row = row - 1 else: machine = next((m for m in machines if m.get("Серійний номер") == serial), {}) ws.cell(row, 1, machine.get("Файл", "")) ws.cell(row, 2, serial) ws.cell(row, 4, machine.get("Етап", "")) ws.cell(row, 9, "Проблем немає") ws.cell(row, 10, "Немає заповнених коментарів із проблемами") start_row = end_row = row row += 1 if end_row > start_row: serial_ranges.append((start_row, end_row)) style_sheet(ws) for start_row, end_row in serial_ranges: ws.merge_cells(start_row=start_row, start_column=2, end_row=end_row, end_column=2) ws.cell(start_row, 2).alignment = Alignment( horizontal="center", vertical="center", wrap_text=True ) machines_ws = wb.create_sheet("Усі серійні номери") machines_ws.append([ "Серійний номер", "Файл", "Етап", "Кількість проблем", "Статус" ]) problem_counts = Counter( clean(r.get("Серійний номер")) for r in records if clean(r.get("Серійний номер")) ) for m in sorted(machines, key=lambda x: clean(x.get("Серійний номер"))): serial = clean(m.get("Серійний номер")) count = problem_counts.get(serial, 0) machines_ws.append([ serial, m.get("Файл", ""), m.get("Етап", ""), count, "Є проблеми" if count else "Проблем немає" ]) style_sheet(machines_ws) # --- Статистика за категоріями --- stats = wb.create_sheet("Статистика") stats.append(["Категорія", "Кількість", "Частка"]) category_counts = Counter(r["Категорія"] for r in records) total = len(records) for category, count in category_counts.most_common(): stats.append([category, count, count / total if total else 0]) stats["E1"] = "Загальна кількість проблем" stats["F1"] = total stats["E2"] = "Оброблено Excel-файлів" stats["F2"] = len(files) stats["E3"] = "Файлів з помилками" stats["F3"] = len(set(e[0] for e in errors)) for cell in stats["C"][1:]: cell.number_format = "0.0%" style_sheet(stats) # --- За етапами --- stage_ws = wb.create_sheet("За етапами") stage_ws.append(["Етап", "Кількість проблем", "Частка"]) stage_counts = Counter(r["Етап"] or "(не вказано)" for r in records) for stage, count in stage_counts.most_common(): stage_ws.append([stage, count, count / total if total else 0]) for cell in stage_ws["C"][1:]: cell.number_format = "0.0%" style_sheet(stage_ws) # --- За тестами --- test_ws = wb.create_sheet("За тестами") test_ws.append(["Етап", "Назва тесту", "Кількість проблем"]) test_counts = Counter((r["Етап"] or "(не вказано)", r["Назва тесту"] or "(не вказано)") for r in records) for (stage, test), count in test_counts.most_common(): test_ws.append([stage, test, count]) style_sheet(test_ws) # --- Повторювані коментарі --- repeat_ws = wb.create_sheet("Повторювані проблеми") repeat_ws.append(["Категорія", "Коментар", "Кількість", "Приклади файлів"]) comment_groups = {} for r in records: key = normalize_comment(r["Коментар"]) if key not in comment_groups: comment_groups[key] = { "text": r["Коментар"], "category": r["Категорія"], "count": 0, "files": [] } comment_groups[key]["count"] += 1 if r["Файл"] not in comment_groups[key]["files"]: comment_groups[key]["files"].append(r["Файл"]) for item in sorted(comment_groups.values(), key=lambda x: (-x["count"], x["text"])): repeat_ws.append([ item["category"], item["text"], item["count"], ", ".join(item["files"][:10]) ]) style_sheet(repeat_ws) # --- Помилки обробки --- if errors: err_ws = wb.create_sheet("Помилки") err_ws.append(["Файл", "Аркуш", "Помилка"]) for err in errors: err_ws.append(list(err)) style_sheet(err_ws) wb.save(output_path) def choose_folder_and_run(): if tk is None: print("Tkinter недоступний. Запустіть програму з терміналу та передайте папку вручну.") folder = input("Шлях до папки з Excel-файлами: ").strip().strip('"') return run_for_folder(folder, None) root = tk.Tk() root.withdraw() folder = filedialog.askdirectory(title="Виберіть папку з Excel-файлами") if not folder: return try: output = run_for_folder(folder, root) messagebox.showinfo( "Готово", f"Звіт створено:\n{output}" ) except Exception as exc: messagebox.showerror("Помилка", f"{exc}\n\n{traceback.format_exc()}") finally: root.destroy() def run_for_folder(folder, root=None): folder = Path(folder) if not folder.exists() or not folder.is_dir(): raise ValueError(f"Папка не існує: {folder}") records, machines, errors, files = process_folder(folder) timestamp = datetime.now().strftime("%Y%m%d_%H%M%S") output_path = folder / f"Статистика_проблем_{timestamp}.xlsx" write_report(records, machines, errors, files, output_path) print(f"Оброблено файлів: {len(files)}") print(f"Знайдено проблем/коментарів: {len(records)}") print(f"Звіт: {output_path}") return str(output_path) if __name__ == "__main__": choose_folder_and_run()
